additive                + -
shift	                << >> >>>
relational	            < > <= >= instanceof
equality	            == !=
bitwise                 AND	&
bitwise exclusive OR	^
bitwise inclusive OR	|
logical AND	            &&
logical OR	            ||
ternary	                ? :
assignment	            = += -= *= /= %= &= ^= |= <<= >>= >>>=
